    \begin{itemize}
        \item Relacher le préhenseur, 
        \item Rentrer les vérins de blocage
    \end{itemize}
     \begin{question}{a5}
        \evaluationProf[2]\\
        Dans la méthode \textit{MExecutionMission} de la classe pince, écrire les lignes de code permettant de réaliser cette mission.\\
    \end{question}
}
